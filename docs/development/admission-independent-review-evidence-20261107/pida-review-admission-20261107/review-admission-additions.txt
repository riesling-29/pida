 const s=await AgentSupervisor.open(options(path,calls,{collaboration:{hostId:'host-1',authorize:()=>true,authorizeAdmission:({phase})=>{if(phase==='dispatch'){dispatch=true;return gate.promise;}return true;}}}));t.after(()=>s.close());
 const root=s.register(owner,{name:'root'}),a=admission(s,root),r=await s.admitOrganization(owner,a);s.start();await until(()=>dispatch);
 await s.cancelOrganization(a.envelope.recipient.org_session_id,owner);gate.resolve(true);await delay(50);const job=s.getJob(r.job_id,owner);assert.equal(job.status,'cancelled');assert.equal(job.runId,undefined);assert.deepEqual(calls,[]);assert.deepEqual(await s.admitOrganization(owner,a),r);assert.equal(s.jobs(root.id,owner).length,1);
